# Flow-matching training deps (in addition to requirements.txt).
#
# INSTALL THE CUDA BUILD THAT MATCHES THE DRIVER, and pin it. A bare
# `pip install --upgrade torch` pulls whatever CUDA build is newest, and if
# that runtime is newer than the installed driver, CUDA tries *forward
# compatibility* -- which is supported only on data-centre GPUs and never on
# GeForce. On a 4090 the result is:
#
#   Error 804: forward compatibility was attempted on non supported HW
#
# and torch.cuda.is_available() is False, so training silently runs on CPU.
# Check `nvidia-smi` for the driver version and pick the wheel accordingly:
#
#   driver >= 580  ->  cu130      driver >= 560  ->  cu126
#   driver >= 570  ->  cu128      driver >= 550  ->  cu124
#
#   pip install "torch==2.6.*" --index-url https://download.pytorch.org/whl/cu124
#
# Pin the version once it works. The upper bound below is deliberate: it is
# there to stop an unrelated `pip install -U` from breaking the cluster again.
torch>=2.2,<3

# Experiment tracking. Optional at runtime: flowmatch/tracking.py no-ops if
# wandb is absent or unauthenticated, so training never dies on a logging
# problem. Needed for the 2x4 grid, where runs are compared by wandb group.
#   wandb login          (or export WANDB_API_KEY=...)
# Air-gapped nodes: pass --wandb-offline and `wandb sync` the run dirs later.
wandb>=0.16
